 function doGet(e) {
  try {
    // Panggil fungsi prosesSurvey dengan parameter dari request
    const hasilSurvey = prosesSurvey(e.parameter);
    
    // Ambil callback name untuk JSONP
    const callback = e.parameter.callback;
    
    // Kembalikan response sebagai JSONP
    return ContentService.createTextOutput(callback + '(' + JSON.stringify(hasilSurvey) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
      
  } catch (error) {
    const callback = e.parameter.callback;
    const response = {
      status: "error",
      message: error.message
    };
    
    return ContentService.createTextOutput(callback + '(' + JSON.stringify(response) + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
}

function doPost(e) {
  try {
    const data = e.parameter;
    
    // Gunakan fungsi prosesSurvey yang sudah ada
    const hasilSurvey = prosesSurvey(data);
    
    // Kembalikan response
    return ContentService.createTextOutput(
      JSON.stringify(hasilSurvey)
    ).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(
      JSON.stringify({
        status: "error",
        message: error.message
      })
    ).setMimeType(ContentService.MimeType.JSON);
  }
}

// Fungsi untuk menghitung semua parameter sekaligus
function hitungParameter(data) {
  const bbIdeal    = calculateBBIdeal(data.mTinggiBadan, data.mUmur);
  const imt        = calculateIMT(data.mBeratBadan, data.mTinggiBadan);
  const lemakTubuh = calculateLemakTubuh(data.mBeratBadan, data.mTinggiBadan, data.mUmur, data.mJenisKelamin);
  const lemakPerut = calculateLemakPerut(data.mLingkarPerut, data.mJenisKelamin, data.mUmur);
  const massaOtot  = calculateMassaOtot(data.mJenisKelamin, lemakTubuh);
  const bmr        = calculateBMR(data.mBeratBadan, data.mTinggiBadan, data.mUmur, data.mJenisKelamin);
  const kalori     = calculateKalori(bmr, data.mAktivitasFisik);
  const protein    = calculateProtein(data.mBeratBadan, data.mJenisKelamin, data.mAktivitasFisik);
  const skor       = Number(calculateSkor(data));
  return [bbIdeal, imt, lemakTubuh, lemakPerut, massaOtot, bmr, kalori, protein, skor];
}

// Untuk menghitung Berat Ideal
function calculateBBIdeal(tinggi, usia) {
  if (usia >= 40) {
    const beratIdeal = tinggi - 100;
    return Number(beratIdeal.toFixed(1));
  } else {
    const beratIdeal = (tinggi - 100) * 0.9;
    return Number(beratIdeal.toFixed(1));
  }
}

// Untuk menghitung IMT / BMI
function calculateIMT(berat, tinggi) {
  const tinggiMeter = tinggi / 100;
  return (berat / (tinggiMeter * tinggiMeter)).toFixed(1);
}

// Untuk menghitung lemak tubuh, menggunakan rumus Siri (1956)
function calculateLemakTubuh(berat, tinggi, umur, jenkel) {
  berat  = Number(berat);
  tinggi = Number(tinggi);
  umur   = Number(umur);
  jenkel = jenkel.trim().charAt(0).toUpperCase() + jenkel.trim().slice(1).toLowerCase(); // "pria" → "Pria"

  if (isNaN(berat) || isNaN(tinggi) || isNaN(umur) || tinggi <= 0) {
    return 0; // Jika input tidak valid
  }

  const tinggiMeter = tinggi / 100;
  const imt = berat / (tinggiMeter * tinggiMeter);

  let lemak;

  if (jenkel === "Pria") {
    lemak = 1.2 * imt + 0.23 * umur - 10.8 - 5.4;
  } else if (jenkel === "Wanita") {
    lemak = 1.2 * imt + 0.23 * umur - 5.4 - 2;
  } else {
    return 0; // Jika jenis kelamin tidak valid
  }

  return lemak.toFixed(0);
}

// Untuk menghitung lemak perut / lingkar perut
function calculateLemakPerut(lingkarPerut) {
  return parseFloat(lingkarPerut);  // hanya return angka lingkar pinggang, dalam cm
}

// Untuk menghitung Massa Otot
function calculateMassaOtot(jenkel, lemakTubuh) {
  lemakTubuh = Number(lemakTubuh);

  if (isNaN(lemakTubuh) || lemakTubuh <= 0 || lemakTubuh >= 100) {
    return 0; // atau return 0; tergantung kamu ingin nilai default apa
  }

  jenkel = jenkel.trim().charAt(0).toUpperCase() + jenkel.trim().slice(1).toLowerCase();
  if (jenkel === "Pria") {
    return ((100 - lemakTubuh) - 10.8).toFixed(0);
  } else if (jenkel === "Wanita") {
    return ((100 - lemakTubuh) - 5.4 - 2).toFixed(0);
  } else {
    return 0  // jenis kelamin tidak dikenali
  }
}

// Untuk menghitung BMR
function calculateBMR(berat, tinggi, umur, jenkel) {  
  if (jenkel === "Pria") {
    return (88.362 + (13.397 * berat) + (4.799 * tinggi) - (5.677 * umur)).toFixed(0);
  } else {
    return (447.593 + (9.247 * berat) + (3.098 * tinggi) - (4.330 * umur)).toFixed(0);
  }
}

// Untuk menghitung kebutuhan kalori berdasarkan BMR dan aktifitas
function calculateKalori(bmr, aktivitas) {
  const faktor = {
    "Sedikit/Tidak Olahraga": 1.2,
    "Olahraga ringan 1-3 hari/minggu": 1.375,
    "Olahraga moderat 3-5 hari/minggu": 1.550,
    "Olahraga berat 6-7 hari per minggu": 1.725,
    "Olahraga sangat aktif + Kegiatan fisik" : 1.9
  };
  return (bmr * (faktor[aktivitas] || 1.2)).toFixed(0);
}

// Untuk menghitung kebutuhan protein berdasarkan aktifitas
function calculateProtein(berat, jenkel, aktifitas) {
  const faktorProtein = {
    "Sedikit/Tidak Olahraga":                 jenkel === "Pria" ? 1.0 : 0.8,
    "Olahraga ringan 1-3 hari/minggu":        jenkel === "Pria" ? 1.2 : 1.0,
    "Olahraga moderat 3-5 hari/minggu":       jenkel === "Pria" ? 1.4 : 1.2,
    "Olahraga berat 6-7 hari per minggu":     jenkel === "Pria" ? 1.6 : 1.4,
    "Olahraga sangat aktif + Kegiatan fisik": jenkel === "Pria" ? 2.0 : 1.8
  };
  const faktor = faktorProtein[aktifitas] || 1.2;  // default moderat
  return (berat * faktor).toFixed(0);              // hasil gram/hari
}

// Untuk menghitung total skor dimana jika 1 = jawaban postif dan jika 0 = jawaban negatif
function calculateSkor(data) {
  const pertanyaan = [
    { field: "mSarapan",  nilai: 1 },
    { field: "mMenu",     nilai: 0 },
    { field: "mNgantuk",  nilai: 0 },
    { field: "mLelah",    nilai: 0 },
    { field: "mLambung",  nilai: 0 },
    { field: "mLapar",    nilai: 0 },
    { field: "mAir",      nilai: 1 },
    { field: "mKonsumsi", nilai: 0 },
    { field: "mBab",      nilai: 0 },
    { field: "mKeram",    nilai: 0 },
    { field: "mTidur",    nilai: 1 },
    { field: "mOlahraga", nilai: 1 },
    { field: "mBak",      nilai: 0 },
    { field: "mKerutan",  nilai: 0 },
    { field: "mSulit",    nilai: 0 },
    { field: "mBBNaik",   nilai: 0 },
    { field: "mRokok",    nilai: 0 },
    { field: "mAlkohol",  nilai: 0 }
  ];

  let skor = 0;
  pertanyaan.forEach(q => {
    const jawaban = data[q.field];
    if ((jawaban === "Ya" && q.nilai === 1) || (jawaban === "Tidak" && q.nilai === 0)) {
      skor += 1;
    }
  });
  return skor;
}

// Fungsi untuk mencari posisi baris terakhir dari suatu kolom
function ambilBrsAkhirKol(sh, posisiKolom) {
  const brsAkhir = sh.getMaxRows();                        // ambil total baris, bukan lastRow berdasarkan semua kolom
  const area     = sh.getRange(posisiKolom + "1:" + posisiKolom + brsAkhir);
  const nilai    = area.getValues();                      // ambil semua isi kolom (array 2D)

  // Scan dari bawah ke atas
  for (let i = nilai.length - 1; i >= 0; i--) {
    if (nilai[i][0] !== "" && nilai[i][0] !== null) {
      const alamatCell  = posisiKolom + (i + 1);          // +1 karena index array mulai dari 0
      const isiCell     = sh.getRange(alamatCell).getValue();
      return isiCell;
    }
  }
  return ""; // kalau tidak ketemu isi
}
